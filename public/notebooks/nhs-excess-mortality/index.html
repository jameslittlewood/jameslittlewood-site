1. Load and check the data¶

Missing observations, duplicate area–sex–period records and invalid confidence limits stop the analysis with an explanation.

def period_start(label):
    """Read either '2022 - 24' or '2022–2024' as a three-year period."""
    match = re.fullmatch(r"(\d{4})\s*[-–]\s*(\d{2}|\d{4})", label)
    if not match:
        raise ValueError(f"Unrecognised period: {label}")
    start = int(match[1])
    end = int(match[2])
    if len(match[2]) == 2:
        end += start // 100 * 100
        if end < start:
            end += 100
    if end != start + 2:
        raise ValueError(f"Expected a three-year period: {label}")
    return start


def check_numbers(rows, description):
    """Require published estimates with valid 95% confidence limits."""
    if rows.empty:
        raise ValueError(f"No observations found: {description}")
    if not np.isfinite(rows[NUMERIC].to_numpy()).all():
        raise ValueError(f"Missing or non-numeric estimate/confidence limit: {description}")
    invalid = (rows[LOWER] > rows[VALUE]) | (rows[VALUE] > rows[UPPER])
    if invalid.any():
        raise ValueError(f"Confidence limits do not bracket the estimate: {description}")


source = CSV_PATH.expanduser()
if not source.is_file():
    raise FileNotFoundError(f"Check CSV_PATH: {source}")
try:
    raw = pd.read_csv(source, dtype=str, keep_default_na=False, encoding="utf-8-sig")
except UnicodeDecodeError:
    raw = pd.read_csv(source, dtype=str, keep_default_na=False, encoding="cp1252")

required = ["Indicator ID", "Area Code", "Area Name", "Area Type",
            "Sex", "Age", "Time period", *NUMERIC]
missing = set(required) - set(raw.columns)
if missing:
    raise ValueError(f"Missing columns: {sorted(missing)}")
for column in raw.columns:
    raw[column] = raw[column].str.strip()

matches = (
    (raw["Indicator ID"] == INDICATOR)
    & (raw["Age"] == AGE)
    & raw["Area Code"].isin(AREAS)
    & raw["Sex"].isin(SEXES)
)
data = raw.loc[matches].copy()
if data.empty:
    raise ValueError("No rows match the indicator, age, areas and sexes.")
for column in ["Category Type", "Category"]:
    if column in data and data[column].ne("").any():
        raise ValueError(f"Unexpected subgroup entries in {column}; headline data required.")

data["Start"] = data["Time period"].map(period_start)
data["Period"] = data["Start"].map(lambda year: f"{year}–{(year + 2) % 100:02d}")
for column in NUMERIC:
    data[column] = pd.to_numeric(data[column].str.replace(",", "", regex=False), errors="coerce")
if data.duplicated(["Area Code", "Sex", "Start"]).any():
    raise ValueError("Duplicate observations for an area, sex and period.")

for area_code, name in AREAS.items():
    area = data[data["Area Code"] == area_code]
    expected_type = "England" if area_code == ENGLAND else (
        "Regions (statistical)" if name == "South East" else "UA"
    )
    if area.empty or not area["Area Name"].str.contains(name, case=False, regex=False).all():
        raise ValueError(f"Missing or inconsistent area: {name}")
    if not area["Area Type"].eq(expected_type).all():
        raise ValueError(f"Unexpected geography type: {name}")

for column in ["Value note", "Compared to England value or percentiles"]:
    if column not in data:
        data[column] = ""
data = data.sort_values(["Start", "Area Code", "Sex"])
persons = data[data["Sex"] == "Persons"].copy()
check_numbers(persons, "Persons comparison data")
coverage = persons.groupby("Area Code")["Start"].apply(set)
if any(coverage.get(area, set()) != set(persons["Start"]) for area in AREAS):
    raise ValueError("All five areas must cover the same periods for the main charts.")
latest_year = persons["Start"].max()
latest_label = persons.loc[persons["Start"] == latest_year, "Period"].iloc[0]
print(f"Loaded {len(data):,} relevant observations. Main comparison period: {latest_label}.")

def trend_chart(show_ci=False):
    fig, ax = plt.subplots(figsize=(10, 4.5), layout="constrained")
    for area_code, colour, marker in [(ENGLAND, GREY, "s"), (READING, PURPLE, "o")]:
        rows = persons[persons["Area Code"] == area_code].sort_values("Start")
        x = np.arange(len(rows))
        ax.plot(x, rows[VALUE], color=colour, marker=marker,
                linestyle="--" if area_code == ENGLAND else "-",
                label=AREAS[area_code], linewidth=2.2, markersize=6)
        if show_ci and area_code == READING:
            ax.fill_between(x, rows[LOWER], rows[UPPER], color=colour, alpha=0.15,
                            label="Reading 95% confidence interval", linewidth=0)
    ax.set_xticks(x, rows["Period"])
    ax.set_ylim(bottom=0)
    ax.set(
           xlabel="Three-year period", ylabel="Excess mortality (%)")
    ax.set_title("Reading and England: excess mortality", loc="left",
                 fontsize=14, fontweight="bold", pad=45)
    ax.grid(axis="y", color="#E7E7E7", linewidth=0.8)
    ax.set_axisbelow(True)
    ax.spines["left"].set_visible(False)
    ax.spines["bottom"].set_color("#BBBBBB")
    ax.tick_params(axis="both", length=0, pad=8)
    ax.legend(loc="lower left", bbox_to_anchor=(0, 1.01),
              ncol=3, frameon=False, fontsize=9, borderaxespad=0)
    plt.show()
    plt.close(fig)

def comparison_chart(rows, labels, title, colours, markers=None):
    """Compact dot-and-whisker chart, sized to the number of rows."""
    if markers is None:
        markers = ["o"] * len(rows)

    height = 1.8 + 0.52 * len(rows)
    fig, ax = plt.subplots(figsize=(10, height))
    fig.subplots_adjust(left=0.18, right=0.81, top=0.76, bottom=0.24)
    fig.suptitle(title, x=0.18, y=0.97, ha="left", fontsize=14, fontweight="bold")
    fig.text(0.18, 0.86, "Dots: estimates   |   Whiskers: 95% confidence intervals",
             fontsize=10, color=GREY)

    for position, (_, row) in enumerate(rows.iterrows()):
        value, lower, upper = row[VALUE], row[LOWER], row[UPPER]
        if labels[position] == "Reading":
            ax.axhspan(position - 0.4, position + 0.4, color="#F2EBF9", zorder=0)
        ax.errorbar(value, position, xerr=[[value-lower], [upper-value]],
                    fmt=markers[position], color=colours[position],
                    capsize=5, markersize=8, elinewidth=1.8, zorder=3)
        # A separate aligned column keeps numbers away from the whiskers.
        ax.text(1.04, position, f"{value:.1f}%",
                transform=ax.get_yaxis_transform(), va="center",
                color=colours[position], fontweight="bold", clip_on=False)

    ax.set_yticks(range(len(rows)), labels)
    ax.set_ylim(len(rows) - 0.35, -0.65)
    ax.set_xlim(min(0, rows[LOWER].min() * 1.1), rows[UPPER].max() * 1.07)
    ax.set_xlabel("Excess mortality (%)", labelpad=10)
    ax.spines["left"].set_visible(False)
    ax.spines["bottom"].set_color("#BBBBBB")
    ax.tick_params(axis="both", length=0, pad=8)
    ax.grid(axis="x", color="#E7E7E7", linewidth=0.8)
    ax.set_axisbelow(True)
    plt.show()
    plt.close(fig)

4. Reading: sex breakdown¶

Male and Female are shown on the chart. Persons is the combined population and appears in the table; it is not the simple average of the two estimates.

The latest Reading period is used, requiring all three sex categories in that period. The male-minus-female difference is descriptive, in percentage points. No formal significance test is performed. Confidence-interval overlap alone does not establish whether the sexes differ significantly.
