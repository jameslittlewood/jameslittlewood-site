fig, ax = plt.subplots(figsize=(10, 5))

x = range(len(df_england))

# reading

ax.fill_between(x, df_reading['Lower CI 95.0 limit'], df_reading['Upper CI 95.0 limit'], color='#5B2C8D', alpha=0.15, label="Reading 95% confidence interval")

# styling
ax.set_title("Reading and England: excess mortality", loc="left", fontsize=14, fontweight="bold", pad=40)
ax.set(xlabel="Period", ylabel="Excess mortality (%)")
ax.grid(axis="y", color="#E7E7E7", linewidth=0.8)
ax.set_xticks(x, df_england['Time period'])
ax.set_ylim(bottom=0)
ax.legend(loc="lower left", bbox_to_anchor=(0, 1.01), ncol=3, frameon=False, fontsize=9, borderaxespad=0);
